    79→export function generateFlights(origin: string, destination: string, count = 8): Flight[] {
    80→  const flights: Flight[] = [];
    81→  const depHours = ["06:00", "07:30", "09:15", "11:00", "13:45", "16:20", "18:30", "20:10", "22:05"];
    82→  for (let i = 0; i < count; i++) {
    83→    const al = pick(AIRLINES, i);
    84→    const dep = pick(depHours, i);
    85→    const dur = 2 + (i % 4) + (i % 2) * 0.5;
    86→    const arrH = (parseInt(dep.slice(0, 2)) + Math.floor(dur)) % 24;
    87→    const arrM = (parseInt(dep.slice(3, 5)) + Math.round((dur % 1) * 60)) % 60;
    88→    const stops = i % 5 === 0 ? 1 : 0;
    89→    flights.push({
    90→      id: `fl-${i + 1}`, airline: al.name, airlineCode: al.code,
    91→      flightNumber: `${al.code}${100 + i * 37}`,
    92→      origin, originCity: pick(CITIES.filter(c => c.code === origin), 0)?.city || "Mumbai",
    93→      destination, destinationCity: pick(CITIES.filter(c => c.code === destination), 0)?.city || "Delhi",
    94→      departTime: dep,
    95→      arriveTime: `${String(arrH).padStart(2, "0")}:${String(arrM).padStart(2, "0")}`,
    96→      duration: `${Math.floor(dur)}h ${Math.round((dur % 1) * 60)}m`,
    97→      stops,
    98→      price: 4200 + (i % 4) * 1800 + (stops ? 1200 : 0),
    99→      currency: "INR",
   100→      cabin: i % 6 === 0 ? "Business" : "Economy",
   101→      seatsLeft: 4 + (i % 12),
   102→      refundable: i % 3 !== 0,
   103→      aircraft: al.aircraft,
   104→      rating: al.rating,
   105→    });
   106→  }
   107→  return flights.sort((a, b) => a.price - b.price);
   108→}
