    26→function StatCard({ icon: Icon, label, value, change, trend, color, subtitle }: {
    27→  icon: React.ElementType; label: string; value: string; change?: number; trend?: "up" | "down"; color: string; subtitle?: string;
    28→}) {
    29→  return (
    30→    <motion.div initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }}>
    31→      <Card className="relative overflow-hidden hover:shadow-md transition-shadow">

    34→            <div className={cn("w-10 h-10 rounded-xl flex items-center justify-center", color)}>
    35→              <Icon className="w-5 h-5" />
    36→            </div>
    37→            {change !== undefined && trend && (
    38→              <span className={cn("text-xs font-medium flex items-center gap-0.5", trend === "up" ? "text-emerald-600" : "text-rose-600")}>
